/** * BACKEND — Cuentas por Cobrar (CxC) — HVAC S.A. de C.V. * Control por contrato: Importe, Cobrado, Facturado (Saldo y Saldo x Facturar se calculan solos) * * Cómo instalarlo (igual que hiciste con el de CxP): * 1. Crea un Google Sheet nuevo (por ejemplo "HVAC - Cuentas por Cobrar"). No uses el mismo * Sheet de CxP ni el de obras — este va aparte. * 2. Extensiones > Apps Script. * 3. Borra el contenido de Code.gs y pega TODO este archivo. * 4. Guarda el proyecto. * 5. Implementar > Nueva implementación > tipo "Aplicación web". * - Ejecutar como: Yo (tu cuenta) * - Quién tiene acceso: Cualquier usuario * 6. Copia la URL que termina en /exec y pégala en el dashboard con el ícono de engrane * "⚙ Conexión". * 7. Si editas este código después, siempre haz "Nueva implementación", no reutilices una vieja. */ const SHEET_NAME = 'Contratos'; const VENTAS_SHEET_NAME = 'PrecioVenta'; const HEADERS = ['ID','Obra','Fecha','Contrato','Moneda','Importe','Cobrado','Facturado']; const VENTA_HEADERS = ['Obra','PrecioVenta','Moneda']; function doGet(e) { return jsonOut({ ok: true, data: readContratos(getSheet()), ventas: readVentas(getVentasSheet()) }); } function doPost(e) { const lock = LockService.getScriptLock(); lock.waitLock(30000); try { const body = JSON.parse(e.postData.contents); const action = body.action; const sheet = getSheet(); if (action === 'add') { appendRow(sheet, body.contrato); } else if (action === 'update') { updateRow(sheet, body.contrato); } else if (action === 'delete') { deleteRow(sheet, body.id); } else if (action === 'bulkImport') { // Solo importa si la hoja está vacía, para no duplicar datos si se llama por error. const existing = readContratos(sheet); if (existing.length === 0 && Array.isArray(body.contratos)) { body.contratos.forEach(c => appendRow(sheet, c)); } } else if (action === 'setVenta') { upsertVenta(getVentasSheet(), body.venta); } return jsonOut({ ok: true, data: readContratos(sheet), ventas: readVentas(getVentasSheet()) }); } catch (err) { return jsonOut({ ok: false, error: String(err) }); } finally { lock.releaseLock(); } } function jsonOut(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); } function getSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow(HEADERS); sheet.setFrozenRows(1); } return sheet; } function readContratos(sheet) { const values = sheet.getDataRange().getValues(); const rows = values.slice(1); return rows.filter(r => r[0]).map(r => ({ id: r[0], obra: r[1], fecha: fmtDate(r[2]), numero: r[3], moneda: r[4], importe: Number(r[5]) || 0, cobrado: Number(r[6]) || 0, facturado: Number(r[7]) || 0 })); } function fmtDate(v) { if (!v) return ''; if (Object.prototype.toString.call(v) === '[object Date]') { return Utilities.formatDate(v, Session.getScriptTimeZone(), 'yyyy-MM-dd'); } return String(v); } function appendRow(sheet, c) { sheet.appendRow([c.id, c.obra, c.fecha, c.numero, c.moneda, c.importe, c.cobrado, c.facturado]); } function findRowIndexById(sheet, id) { const values = sheet.getDataRange().getValues(); for (let i = 1; i < values.length; i++) { if (String(values[i][0]) === String(id)) return i + 1; } return -1; } function updateRow(sheet, c) { const rowIdx = findRowIndexById(sheet, c.id); if (rowIdx === -1) return; sheet.getRange(rowIdx, 1, 1, HEADERS.length).setValues([[ c.id, c.obra, c.fecha, c.numero, c.moneda, c.importe, c.cobrado, c.facturado ]]); } function deleteRow(sheet, id) { const rowIdx = findRowIndexById(sheet, id); if (rowIdx === -1) return; sheet.deleteRow(rowIdx); } // --- Precio de venta por obra (total único, sin desglose) --- function getVentasSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(VENTAS_SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(VENTAS_SHEET_NAME); sheet.appendRow(VENTA_HEADERS); sheet.setFrozenRows(1); } return sheet; } function readVentas(sheet) { const values = sheet.getDataRange().getValues(); const rows = values.slice(1); return rows.filter(r => r[0]).map(r => ({ obra: r[0], precioVenta: Number(r[1]) || 0, moneda: r[2] || 'MXN' })); } function findVentaRowIndex(sheet, obra) { const values = sheet.getDataRange().getValues(); for (let i = 1; i < values.length; i++) { if (String(values[i][0]) === String(obra)) return i + 1; } return -1; } function upsertVenta(sheet, v) { if (!v || !v.obra) return; const row = [v.obra, Number(v.precioVenta) || 0, v.moneda || 'MXN']; const rowIdx = findVentaRowIndex(sheet, v.obra); if (rowIdx === -1) { sheet.appendRow(row); } else { sheet.getRange(rowIdx, 1, 1, VENTA_HEADERS.length).setValues([row]); } }
