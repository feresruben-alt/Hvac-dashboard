/** * BACKEND — Cuentas por Pagar (CxP) — HVAC S.A. de C.V. * * Cómo instalarlo: * 1. Crea un Google Sheet nuevo (puede llamarse "HVAC - Cuentas por Pagar"). * 2. Extensiones > Apps Script. * 3. Borra el contenido de Code.gs y pega TODO este archivo. * 4. Guarda el proyecto. * 5. Implementar > Nueva implementación > tipo "Aplicación web". * - Ejecutar como: Yo (tu cuenta) * - Quién tiene acceso: Cualquier usuario <-- IMPORTANTE, igual que el dashboard maestro * 6. Copia la URL de implementación (termina en /exec) y pégala en el dashboard * cuando te la pida (ícono de engrane arriba a la derecha). * 7. Cada vez que edites este código, vuelve a hacer "Nueva implementación" * en lugar de reutilizar una vieja (evita confusión con implementaciones duplicadas). */ const SHEET_NAME = 'Facturas'; const CONFIG_SHEET_NAME = 'Config'; const PRESUPUESTOS_SHEET_NAME = 'Presupuestos'; const HEADERS = ['ID','Fecha','Proveedor','Factura','Obra','Importe','Moneda','Pagada','FechaPago']; const PRESUPUESTO_HEADERS = ['Obra','Equipos','Materiales','ManoObra','Indirecto','Moneda']; function doGet(e) { const action = (e && e.parameter && e.parameter.action) || 'list'; if (action === 'config') { return jsonOut({ ok: true, tc: getTC() }); } return jsonOut({ ok: true, data: readInvoices(getSheet()), presupuestos: readPresupuestos(getPresupuestosSheet()), tc: getTC() }); } function doPost(e) { const lock = LockService.getScriptLock(); lock.waitLock(30000); try { const body = JSON.parse(e.postData.contents); const action = body.action; const sheet = getSheet(); if (action === 'add') { const inv = body.invoice; sheet.appendRow([ inv.id, inv.fecha, inv.proveedor, inv.factura, inv.obra, inv.importe, inv.moneda, inv.pagada ? 'TRUE' : 'FALSE', inv.fechaPago || '' ]); } else if (action === 'update') { updateRow(sheet, body.invoice); } else if (action === 'pay') { setPaid(sheet, body.id, true, body.fechaPago); } else if (action === 'unpay') { setPaid(sheet, body.id, false, ''); } else if (action === 'delete') { deleteRow(sheet, body.id); } else if (action === 'setTC') { setTC(body.tc); } else if (action === 'setPresupuesto') { upsertPresupuesto(getPresupuestosSheet(), body.presupuesto); } return jsonOut({ ok: true, data: readInvoices(sheet), presupuestos: readPresupuestos(getPresupuestosSheet()), tc: getTC() }); } catch (err) { return jsonOut({ ok: false, error: String(err) }); } finally { lock.releaseLock(); } } function jsonOut(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } function getSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow(HEADERS); sheet.setFrozenRows(1); } return sheet; } function getConfigSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(CONFIG_SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(CONFIG_SHEET_NAME); sheet.getRange('A1').setValue('TC_USD_MXN'); sheet.getRange('B1').setValue(17); } return sheet; } function getTC() { const sheet = getConfigSheet(); const v = sheet.getRange('B1').getValue(); return Number(v) || 17; } function setTC(value) { const sheet = getConfigSheet(); sheet.getRange('B1').setValue(Number(value) || 17); } function readInvoices(sheet) { const values = sheet.getDataRange().getValues(); const rows = values.slice(1); return rows.filter(r => r[0]).map(r => ({ id: r[0], fecha: fmtDate(r[1]), proveedor: r[2], factura: r[3], obra: r[4], importe: Number(r[5]) || 0, moneda: r[6], pagada: String(r[7]).toUpperCase() === 'TRUE', fechaPago: r[8] ? fmtDate(r[8]) : '' })); } function fmtDate(v) { if (!v) return ''; if (Object.prototype.toString.call(v) === '[object Date]') { return Utilities.formatDate(v, Session.getScriptTimeZone(), 'yyyy-MM-dd'); } return String(v); } function findRowIndexById(sheet, id) { const values = sheet.getDataRange().getValues(); for (let i = 1; i < values.length; i++) { if (String(values[i][0]) === String(id)) return i + 1; // 1-indexed sheet row } return -1; } function updateRow(sheet, inv) { const rowIdx = findRowIndexById(sheet, inv.id); if (rowIdx === -1) return; sheet.getRange(rowIdx, 1, 1, HEADERS.length).setValues([[ inv.id, inv.fecha, inv.proveedor, inv.factura, inv.obra, inv.importe, inv.moneda, inv.pagada ? 'TRUE' : 'FALSE', inv.fechaPago || '' ]]); } function setPaid(sheet, id, paid, fechaPago) { const rowIdx = findRowIndexById(sheet, id); if (rowIdx === -1) return; sheet.getRange(rowIdx, 8).setValue(paid ? 'TRUE' : 'FALSE'); sheet.getRange(rowIdx, 9).setValue(fechaPago || ''); } function deleteRow(sheet, id) { const rowIdx = findRowIndexById(sheet, id); if (rowIdx === -1) return; sheet.deleteRow(rowIdx); } // --- Presupuesto por obra (costo de la obra: Equipos / Materiales / Mano de obra / Indirecto) --- function getPresupuestosSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(PRESUPUESTOS_SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(PRESUPUESTOS_SHEET_NAME); sheet.appendRow(PRESUPUESTO_HEADERS); sheet.setFrozenRows(1); } return sheet; } function readPresupuestos(sheet) { const values = sheet.getDataRange().getValues(); const rows = values.slice(1); return rows.filter(r => r[0]).map(r => ({ obra: r[0], equipos: Number(r[1]) || 0, materiales: Number(r[2]) || 0, manoObra: Number(r[3]) || 0, indirecto: Number(r[4]) || 0, moneda: r[5] || 'MXN' })); } function findPresupuestoRowIndex(sheet, obra) { const values = sheet.getDataRange().getValues(); for (let i = 1; i < values.length; i++) { if (String(values[i][0]) === String(obra)) return i + 1; } return -1; } function upsertPresupuesto(sheet, p) { if (!p || !p.obra) return; const row = [p.obra, Number(p.equipos) || 0, Number(p.materiales) || 0, Number(p.manoObra) || 0, Number(p.indirecto) || 0, p.moneda || 'MXN']; const rowIdx = findPresupuestoRowIndex(sheet, p.obra); if (rowIdx === -1) { sheet.appendRow(row); } else { sheet.getRange(rowIdx, 1, 1, PRESUPUESTO_HEADERS.length).setValues([row]); } }
